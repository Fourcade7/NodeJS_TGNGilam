const fs = require("fs");
const USERS_FILE = "./users.json";

function checkPhoneNumber(bot) {

  bot.on("message", (msg) => {
    const chatId = msg.chat.id;
    const name = msg.chat.first_name;
    const text = msg.text;

    // Faylni o‘qish
    let users = [];
    if (fs.existsSync(USERS_FILE)) {
      users = JSON.parse(fs.readFileSync(USERS_FILE, "utf-8"));
    }

    // Agar foydalanuvchi oldin ro‘yxatdan o‘tgan bo‘lsa – qayta so‘ramaymiz
    const existingUser = users.find(u => u.chatId === chatId);
    if (existingUser) return;

    // Telefon tekshiruvi
    const phoneRegex = /^\+998\d{9}$/;

    if (!phoneRegex.test(text)) {
      bot.sendMessage(chatId, "📞 Telefon raqamingizni yuboring.\nMasalan: +998901234567");
      return;
    }

    // Agar telefon to‘g‘ri bo‘lsa
    users.push({
      chatId,
      first_name: name,
      phoneNumber: text
    });

    fs.writeFileSync(USERS_FILE, JSON.stringify(users, null, 2));

    bot.sendMessage(chatId, "✅ Telefon raqamingiz muvaffaqiyatli saqlandi!");
    console.log(`Yangi foydalanuvchi: ${name} (${chatId})`);
  });
}
